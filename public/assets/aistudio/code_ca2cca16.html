Embed Socket Client in 4 Lines

Initialize non-blocking socket listeners and stream pipelines with full connection lifecycle management.

client := pulsetcp.NewClient("0.0.0.0:8080", "@alice_core")
client.OnMessage(func(src, msg string) {
    fmt.Printf("[%s]: %s\n", src, msg)
})
client.Broadcast("Hello cluster")
client.Whisper("@bob_worker", "Task #42 complete")
let mut client = PulseClient::connect("0.0.0.0:8080", "@alice_core").await?;
client.listen(|sender, msg| {
    println!("[{sender}] {msg}");
});
client.broadcast("Hello cluster").await?;
client.whisper("@bob_worker", "Task #42 complete").await?;
